
import java.awt.*;

public class SimpleAWTFrame {
    public static void main(String[] args) {
        Frame f = new Frame("My AWT Frame");

        Label l = new Label("Welcome to Java GUI");

        l.setBounds(80, 100, 200, 30);

        f.add(l);

        f.setSize(350, 250);
        f.setLayout(null);
        f.setVisible(true);
    }
}
